Đồng Bộ Bảng Điểm Tức Thì Về Google Sheets

Mỗi khi học sinh nộp bài, điểm số và chi tiết sẽ được tự động ghi vào Google Sheets

Dán URL triển khai dưới dạng Web App (Ứng dụng web) từ Google Apps Script của Thầy/Cô.

Hướng Dẫn Thiết Lập Google Apps Script (Chỉ mất 30 giây):

  1. Tạo một file Google Sheets (Trang tính) mới trên Google Drive của Thầy/Cô.
  2. Trên thanh menu, chọn Tiện ích mở rộng (Extensions) → Apps Script.
  3. Xóa mã cũ, dán đoạn mã bên dưới vào và bấm Lưu (Ctrl + S).
  4. Bấm nút Triển khai (Deploy) → Tùy chọn triển khai mới (New deployment):
    • Chọn loại: Ứng dụng web (Web app)
    • Ai có quyền truy cập (Who has access): Bất kỳ ai (Anyone)
  5. Bấm Triển khai → Sao chép URL ứng dụng web và dán vào ô bên trên.
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Thời gian", "Mã Nhiệm Vụ", "Lớp", "Họ và Tên", "Điểm Số", "Số Câu Đúng", "Thời Gian Làm (giây)", "Đánh Giá AI"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(),
    data.taskId || "",
    data.className || "",
    data.studentName || "",
    data.score || 0,
    (data.correctCount || 0) + "/" + (data.totalQuestions || 0),
    data.timeSpentSeconds || 0,
    data.aiFeedback || ""
  ]);
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
